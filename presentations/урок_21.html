Как Python находит модуль

Три шага поиска при каждом import

1️⃣

Текущая папка

Python смотрит, нет ли файла с таким именем рядом с твоим скриптом. Поэтому не называй файлы random.py или math.py!

2️⃣

Стандартная библиотека

Встроенные модули Python — math, random, time, os. Установлены вместе с Python, ничего скачивать не нужно.

3️⃣

site-packages (pip)

Внешние пакеты, установленные командой pip install. Здесь живут pygame, requests, pillow.

Важно: если Python нашёл модуль на шаге 1 — он там и остановится. Остальные шаги пропускает!

Две формы import

С префиксом или без — зависит от ситуации

import module

Импортируешь весь модуль. Функции вызываются через точку.

import random
import math

random.randint(1, 6)
math.sqrt(16)
# Понятно, откуда функция

Рекомендуется — код читаем

from module import func

Импортируешь только одну функцию. Префикс не нужен.

from random import randint, choice
from math import sqrt, pi

randint(1, 6)   # без random.
sqrt(16)        # без math.

Осторожно: может быть конфликт имён

Правило EdusonGames: импорты — всегда в самом начале файла, до любого кода.

Мини-проект: «Броски костей»

random + math + функции = статистика бросков

Python
import random import math def roll_dice(n, sides=6): """Бросает n кубиков с sides гранями. Возвращает список.""" return [random.randint(1, sides) for _ in range(n)] def dice_stats(rolls): """Статистика бросков.""" total = sum(rolls) avg = total / len(rolls) sixes = rolls.count(6) return total, round(avg, 2), sixes # Игровая сессия print("=== БРОСАЕМ КОСТИ ===") n = 10 results = roll_dice(n) print(f"Результаты ({n} бросков): {results}") total, avg, sixes = dice_stats(results) print(f"Сумма: {total}") print(f"Среднее: {avg} (теоретическое: 3.5)") print(f"Шестёрок: {sixes} из {n}") # Гистограмма print("\nЧастота выпадений:") for face in range(1, 7): count = results.count(face) bar = "█" * count print(f" {face}: {bar} ({count})")

Расширение: сравниваем серии

Несколько игроков бросают кости — кто выиграл?

Python
import random import math def roll_and_sum(n=5, sides=6): """Бросает n кубиков и возвращает сумму и броски.""" rolls = [random.randint(1, sides) for _ in range(n)] return sum(rolls), rolls игроки = ["Анна", "Борис", "Вера"] результаты = {} print("=== ТУРНИР КУБИКОВ ===") for игрок in игроки: сумма, броски = roll_and_sum(5) результаты[игрок] = сумма print(f"{игрок}: {броски} → сумма = {сумма}") print() победитель = max(результаты, key=результаты.get) max_score = max(результаты.values()) min_score = min(результаты.values()) print(f"Победитель: {победитель} с результатом {max_score}!") print(f"Разница: {max_score - min_score} очков") avg = sum(результаты.values()) / len(результаты) print(f"Среднее по всем: {avg:.1f}")

import random import math import time def attack(base_dmg, crit_chance=0.2): """Рассчитывает удар: возможен критический урон.""" roll = random.random() if roll < crit_chance: dmg = math.floor(base_dmg * 1.5) return dmg, True else: dmg = random.randint(base_dmg - 5, base_dmg + 5) return dmg, False hero_hp = 100 boss_hp = 150 round_num = 0 print("=== БОЙ НАЧАЛСЯ ===") while hero_hp > 0 and boss_hp > 0 and round_num < 20: round_num += 1 print(f"\n-- Раунд {round_num} --") # Герой атакует dmg, crit = attack(20) boss_hp -= dmg crit_text = " КРИТ!" if crit else "" print(f"Герой бьёт: -{dmg} HP боссу{crit_text}") if boss_hp <= 0: break # Босс атакует dmg_b, crit_b = attack(15, 0.15) hero_hp -= dmg_b crit_b_text = " КРИТ!" if crit_b else "" print(f"Босс бьёт: -{dmg_b} HP герою{crit_b_text}") print(f"Герой: {max(hero_hp,0)} HP | Босс: {max(boss_hp,0)} HP") time.sleep(0.15) print() if hero_hp > 0 and boss_hp <= 0: print(f"ПОБЕДА! Герой выжил с {hero_hp} HP!") elif boss_hp > 0 and hero_hp <= 0: print("ПОРАЖЕНИЕ. Босс побеждает!") else: print("Ничья — бой затянулся!")

Разбор симулятора: три паттерна

Как три модуля работают вместе

random — шанс крита

roll = random.random()
# 0.0 до 1.0
if roll < 0.2:
    # крит! (20% шанс)

random.random() даёт число от 0 до 1. Сравниваем с порогом — получаем шанс события.

math — расчёт урона

dmg = math.floor(
    base_dmg * 1.5
)
# floor убирает дробь

math.floor() нужен, чтобы урон был целым числом — в играх нет 17.5 HP!

time — драматизм

time.sleep(0.2)
# пауза между раундами
# делает бой "живым"

time.sleep() — пауза в секундах. 0.2 = 200 мс. Создаёт эффект анимации.
