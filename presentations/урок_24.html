# Зачем strip() перед split() lines = [ "Столица России?|Москва|Питер|Казань|Новгород|1\n", # \n в конце " 2**10 = ?|512|1024|256|2048|2 ", # пробелы вокруг "", # пустая строка "def создаёт?|Переменную|Функцию|Список|Цикл|2", # нормальная ] print("=== С strip() ===") for line in lines: line = line.strip() if not line: print(" (пустая — пропускаем)") continue parts = line.split("|") print(f" {len(parts)} полей: '{parts[0]}' → ответ {parts[-1]}") print("\n=== Без strip() — проблемы! ===") line = "Вопрос|A|B|C|D|2\n" parts = line.split("|") print(f"Последний: '{parts[-1]}'") # '2\n' — не число!

strip() убирает

  • "\n" — перенос строки (конец файла)
  • "\r" — возврат каретки (Windows файлы)
  • " " — пробелы по краям
  • "\t" — табуляцию

split("|") vs split()

  • split("|") — делит по трубе, сохраняет пустые
  • split() без аргумента — делит по всем пробелам
  • Для формата pipe всегда используй split("|")
int(parts[-1]) упадёт с ValueError если в конце есть \n. Всегда делай strip() первым!

shuffle() vs sample()

  • shuffle(list) — перемешивает на месте, возвращает None
  • sample(list, k) — возвращает k элементов, список не меняет
  • Для квиза: shuffle() потом срез [:k]

Почему случайность важна

  • Квиз каждый раз другой — нельзя выучить порядок
  • Честное соревнование у разных игроков
  • Из 100 вопросов можно делать разные квизы по 10

Словарь урока — ключевые термины

SRP Single Responsibility Principle — каждая функция делает ровно одно дело
pipe-format Формат данных: поля разделены символом | — удобно для CSV-подобных файлов
__name__ Специальная переменная Python: "__main__" при прямом запуске, имя модуля при импорте
main guard Паттерн if __name__ == "__main__": — защищает код от случайного запуска при импорте
enumerate() Встроенная функция: возвращает (индекс, элемент) при обходе списка
sorted() + lambda Сортировка по ключу: sorted(lst, key=lambda x: x[1], reverse=True)
time.time() Текущее время в секундах с 01.01.1970 (Unix timestamp). Разность = длительность
random.shuffle() Перемешивает список на месте (in-place). Возвращает None!

Что дальше — Модуль 7: ООП

Объектно-ориентированное программирование

  • Классы и объекты — объединяем данные и функции
  • __init__ — конструктор, создаёт объект
  • self — ссылка на текущий объект
  • Методы — функции внутри класса
  • Наследование — код без дублирования
  • Проект: RPG-игра с Hero и Enemy

Вот как это выглядит:

class Hero:
    def __init__(self, name):
        self.name = name
        self.hp = 100
        self.score = 0

    def attack(self, enemy):
        dmg = 20
        enemy.hp -= dmg
        self.score += dmg
        print(f"{self.name} наносит {dmg}!")

player = Hero("Аня")
boss = Hero("Дракон")
player.attack(boss)
print(f"HP дракона: {boss.hp}")
Классы — это способ организовать функции и данные вместе. Вы уже готовы!
