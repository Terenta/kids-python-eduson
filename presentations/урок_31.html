Базовый класс Character

character.py
class Character: def __init__(self, name, hp, damage): self.name = name self.hp = hp self.max_hp = hp self.damage = damage def attack(self, other): other.hp -= self.damage print(f"{self.name} -> {other.name}: -{self.damage} HP") def is_alive(self): return self.hp > 0 def status(self): bars = max(0, self.hp * 10 // self.max_hp) bar = "#" * bars + "." * (10 - bars) print(f"{self.name}: [{bar}] {max(0,self.hp)}/{self.max_hp}") # Тест hero = Character("Артём", 100, 15) goblin = Character("Гоблин", 40, 8) hero.attack(goblin) goblin.status() print("Гоблин жив?", goblin.is_alive())

Методы класса

  • __init__() — name, hp, max_hp, damage
  • attack(other) — уменьшает other.hp
  • is_alive() — возвращает bool
  • status() — ASCII полоска HP
is_alive() — первый пример метода с return.
Использование: while hero.is_alive() and enemy.is_alive():

Класс Hero — игрок

hero.py
class Character: def __init__(self, name, hp, damage): self.name = name self.hp = hp self.max_hp = hp self.damage = damage def attack(self, other): other.hp -= self.damage class Hero(Character): def __init__(self, name, hp, damage): super().__init__(name, hp, damage) self.score = 0 self._special_used = False def heal(self, amount=20): old_hp = self.hp self.hp = min(self.hp + amount, self.max_hp) healed = self.hp - old_hp print(f"{self.name} восстановил {healed} HP") def special_attack(self, other): if not self._special_used: dmg = self.damage * 2 other.hp -= dmg self._special_used = True print(f"СПЕЦАТАКА! {self.name} -> {other.name}: -{dmg} HP!") else: print("Спецатака уже использована!") def win(self): self.score += 1 print(f"{self.name} победил! Счёт: {self.score}") hero = Hero("Артём", 80, 15) enemy = Character("Гоблин", 40, 8) hero.special_attack(enemy) print(f"HP Гоблина: {enemy.hp}") hero.heal(15) print(f"HP героя: {hero.hp}")

min() для ограничения

min(hp + 20, max_hp) — берём меньшее из двух. Не даёт HP выйти за максимум.

_special_used

Приватный флаг — спецатака доступна только раз. Инкапсуляция в действии!

win() → score

Вызываем при победе — счёт растёт. В итоге сравниваем score со списком врагов.

Класс Monster — враг

monster.py
class Character: def __init__(self, name, hp, damage): self.name = name self.hp = hp self.max_hp = hp self.damage = damage def attack(self, other): other.hp -= self.damage print(f"{self.name} -> {other.name}: -{self.damage} HP") class Monster(Character): def __init__(self, name, hp, damage, monster_type="обычный"): super().__init__(name, hp, damage) self.monster_type = monster_type self._base_damage = damage def attack(self, other): if self.hp < self.max_hp * 0.3: self.damage = self._base_damage * 2 print(f"{self.name} в ЯРОСТИ!") else: self.damage = self._base_damage super().attack(other) def describe(self): print(f"{self.name} [{self.monster_type}] HP:{self.hp} DMG:{self.damage}") hero = Character("Артём", 100, 15) goblin = Monster("Гоблин", 40, 8, "быстрый") goblin.describe() goblin.hp = 8 # ярость (< 30% от 40) goblin.attack(hero) print(f"HP героя: {hero.hp}")

Ярость при <30% HP

self.hp < self.max_hp * 0.3 — если меньше 30% — удваиваем урон. Динамическая сложность!

super().attack(other)

Переопределяем attack(), но вызываем родительский в конце — не дублируем код.

Типы монстров:
Гоблин — быстрый (малый HP, большой урон)
Орк — танк (много HP, средний урон)
Дракон — босс (много всего)

return_methods.py
class Hero: def __init__(self, name, hp): self.name = name self.hp = hp self.max_hp = hp def is_alive(self): return self.hp > 0 def heal(self, amount): old = self.hp self.hp = min(self.hp + amount, self.max_hp) return self.hp - old # сколько вылечили def get_info(self): return f"{self.name}: {self.hp} HP" hero = Hero("Артём", 60) print("Жив?", hero.is_alive()) healed = hero.heal(20) print(f"Вылечено: {healed} HP") info = hero.get_info() print(info)

Типы возвращаемых значений

  • return True/False — bool для условий
  • return число — int/float для вычислений
  • return строка — str для вывода
  • return список — list для коллекций
Без return — None!
Если метод ничего не возвращает — Python возвращает None.
result = hero.attack(enemy) → result == None

Задание: банковский счёт

Класс Account

  1. __init__(owner, balance=0)
  2. deposit(amount) → возвращает новый баланс
  3. withdraw(amount) → если хватает — снимаем; иначе "Недостаточно средств"
  4. transfer(other, amount) → снять у себя, зачислить другому

Бонус

Добавь __history = [] (приватный список). В deposit/withdraw добавляй строку в историю. Метод show_history() печатает все операции.

account.py
class Account: def __init__(self, owner, balance=0): self.owner = owner self._balance = balance self.__history = [] def deposit(self, amount): self._balance += amount self.__history.append(f"+{amount}") return self._balance def withdraw(self, amount): if amount <= self._balance: self._balance -= amount self.__history.append(f"-{amount}") return self._balance else: print("Недостаточно средств!") return self._balance def transfer(self, other, amount): old = self._balance self.withdraw(amount) if self._balance < old: other.deposit(amount) print(f"Перевод {amount} -> {other.owner}") def get_balance(self): return self._balance def show_history(self): print(f"История {self.owner}:", self.__history) alice = Account("Алиса", 500) bob = Account("Боб", 100) alice.deposit(200) alice.withdraw(100) alice.transfer(bob, 150) print(f"Алиса: {alice.get_balance()}") print(f"Боб: {bob.get_balance()}") alice.show_history()

Класс Battle — пошаговый бой

import random class Fighter: def __init__(self, name, hp, atk, defense=0): self.name = name self.hp = hp self.atk = atk self.defense = defense def is_alive(self): return self.hp > 0 def __str__(self): return f"{self.name} (HP:{max(0,self.hp)})" class Battle: def __init__(self, fighter1, fighter2): self.f1 = fighter1 self.f2 = fighter2 self.round = 0 def do_round(self): self.round += 1 print(f"--- Раунд {self.round} ---") self._hit(self.f1, self.f2) if self.f2.is_alive(): self._hit(self.f2, self.f1) def _hit(self, attacker, target): dmg = max(1, attacker.atk - target.defense + random.randint(-3,3)) target.hp -= dmg print(f" {attacker.name} → {target.name}: -{dmg} | {target}") def run(self): print(f"⚔️ {self.f1.name} vs {self.f2.name}!\n") while self.f1.is_alive() and self.f2.is_alive(): self.do_round() winner = self.f1 if self.f1.is_alive() else self.f2 print(f"\n🏆 Победитель: {winner.name}!") hero = Fighter("Артур", 100, 35, 8) enemy = Fighter("Дракон", 120, 40, 5) Battle(hero, enemy).run()

HP: {self.hp}") def __str__(self): return f"{self.name} [Ур.{self.level}] HP:{self.hp} XP:{self.xp}/{self.xp_to_next}" hero = Hero("Артур") print(hero) hero.gain_xp(60) hero.gain_xp(70) # level up! hero.gain_xp(80) hero.gain_xp(120) # может быть double level up print(hero)

import random class Item: def __init__(self, name, bonus): self.name = name; self.bonus = bonus class Hero: def __init__(self, name, hp, atk): self.name = name; self.hp = hp; self.atk = atk; self.xp = 0 self.items = [] def add_item(self, item): self.items.append(item) @property def total_atk(self): return self.atk + sum(i.bonus for i in self.items) def is_alive(self): return self.hp > 0 def gain_xp(self, n): self.xp += n class Enemy: def __init__(self, name, hp, atk, xp_reward): self.name = name; self.hp = hp; self.atk = atk; self.xp_reward = xp_reward def is_alive(self): return self.hp > 0 def battle(hero, enemy): print(f"\n⚔️ {hero.name} vs {enemy.name}!") rd = 0 while hero.is_alive() and enemy.is_alive(): rd += 1 dmg_h = hero.total_atk + random.randint(-5, 5) enemy.hp = max(0, enemy.hp - dmg_h) dmg_e = enemy.atk + random.randint(-3, 3) hero.hp = max(0, hero.hp - dmg_e) if rd <= 3: print(f" Р{rd}: {hero.name} -{dmg_h} → {enemy.name}(HP:{enemy.hp}) | {enemy.name} -{dmg_e} → {hero.name}(HP:{hero.hp})") if hero.is_alive(): hero.gain_xp(enemy.xp_reward) print(f" 🏆 {hero.name} победил! +{enemy.xp_reward} XP (итого {hero.xp})") else: print(f" 💀 {hero.name} побеждён!") hero = Hero("Артур", 150, 35) hero.add_item(Item("Меч рыцаря", 15)) enemies = [Enemy("Гоблин", 50, 15, 30), Enemy("Тролль", 90, 25, 70), Enemy("Дракон", 140, 40, 150)] for e in enemies: battle(hero, e) print(f"\nИтог: {hero.name} | HP:{hero.hp} | XP:{hero.xp}")
