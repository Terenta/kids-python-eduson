Схема Тамагочи

🍖 feed()

hunger -= 3, energy += 1. Нельзя ниже 0

🎾 play()

happiness += 5, energy -= 2, hunger += 1

💤 sleep()

energy += 5, happiness += 1. tick() не идёт

class Pet: def __init__(self, name): self.name = name self.hunger = 3 # 0=сыт, 10=очень голодный self.happiness = 7 # 0=грустный, 10=счастливый self.energy = 8 # 0=без сил, 10=бодрый self.age = 0 # ходы жизни def _clamp(self, val): return max(0, min(10, val)) def feed(self): self.hunger = self._clamp(self.hunger - 3) self.energy = self._clamp(self.energy + 1) print(f"{self.name} поел! Голод: {self.hunger}") def play(self): if self.energy < 2: print(f"{self.name} слишком устал для игры!") return self.happiness = self._clamp(self.happiness + 5) self.energy = self._clamp(self.energy - 2) self.hunger = self._clamp(self.hunger + 1) print(f"{self.name} поиграл! Счастье: {self.happiness}") def sleep(self): self.energy = self._clamp(self.energy + 5) self.happiness = self._clamp(self.happiness + 1) print(f"{self.name} поспал! Энергия: {self.energy}") def tick(self): self.hunger = self._clamp(self.hunger + 1) self.happiness = self._clamp(self.happiness - 1) self.energy = self._clamp(self.energy - 1) self.age += 1 def is_alive(self): return self.hunger < 10 and self.energy > 0 def status(self): def bar(v): return "#"*v + "."*(10-v) print(f"\n=== {self.name} (возраст: {self.age}) ===") print(f" Голод: [{bar(self.hunger)}] {self.hunger}/10") print(f" Счастье: [{bar(self.happiness)}] {self.happiness}/10") print(f" Энергия: [{bar(self.energy)}] {self.energy}/10") # Тест p = Pet("Пушок") p.status() p.feed() p.play() p.tick() p.status() print("Живой?", p.is_alive())

Cat и Dog — особенности

cat_dog.py
class Pet: def __init__(self, name): self.name = name self.hunger = 3 self.happiness = 7 self.energy = 8 self.age = 0 def _clamp(self, v): return max(0, min(10, v)) def feed(self): self.hunger = self._clamp(self.hunger - 3) print(f"{self.name} поел!") def play(self): self.happiness = self._clamp(self.happiness + 5) self.energy = self._clamp(self.energy - 2) print(f"{self.name} поиграл!") class Cat(Pet): def play(self): self.happiness = self._clamp(self.happiness + 3) self.energy = self._clamp(self.energy - 1) print(f"{self.name} немного поиграла...") def purr(self): if self.happiness >= 7: print(f"Мурр... {self.name} мурчит!") else: print(f"{self.name} не в настроении...") class Dog(Pet): def feed(self): super().feed() self.hunger = self._clamp(self.hunger + 2) print(f" {self.name} смотрит умоляюще ещё раз...") def fetch(self): self.happiness = self._clamp(self.happiness + 4) self.energy = self._clamp(self.energy - 3) print(f"{self.name} принёс мяч! Счастье: {self.happiness}") cat = Cat("Мурка") dog = Dog("Шарик") cat.play() cat.purr() dog.feed() dog.fetch()

🐱 Cat — ленивая кошка

  • play() +3 счастья (не +5)
  • play() -1 энергия (не -2)
  • purr() если happiness ≥ 7

🐶 Dog — прожорливая

  • feed() вызывает super().feed()
  • После еды hunger += 2
  • fetch() +4 счастья, -3 энергия

ASCII_CAT = { "happy": " /\\_/\\\n ( ^.^ )\n > w <\n /| |\\", "sad": " /\\_/\\\n ( -.- )\n > . <\n /| |\\", "hungry": " /\\_/\\\n ( x.x )\n > ~ <\n /| |\\", "tired": " /\\_/\\\n ( -.-) \n > z <\n /| |\\", } def get_cat_art(happiness, hunger, energy): if hunger >= 8: return ASCII_CAT["hungry"] elif energy <= 2: return ASCII_CAT["tired"] elif happiness >= 7: return ASCII_CAT["happy"] else: return ASCII_CAT["sad"] # Тест print("Голодная кошка:") print(get_cat_art(5, 9, 5)) print("\nСчастливая кошка:") print(get_cat_art(9, 2, 8))

class Pet:

def game(pet_name, pet_class=Pet):
    pet = pet_class(pet_name)
    print(f"Добро пожаловать! Твой питомец: {pet.name}")

    while True:
        pet.status()
        if not pet.is_alive():
            print(f"{pet.name} нуждается в уходе!")
            break

        print("\nЧто сделать?")
        print("  1 - Покормить")
        print("  2 - Поиграть")
        print("  3 - Поспать")
        print("  4 - Выйти")

        choice = input("Ввод (1-4): ").strip()

        if choice == "1":
            pet.feed()
        elif choice == "2":
            pet.play()
        elif choice == "3":
            pet.sleep()
        elif choice == "4":
            print(f"Пока! {pet.name} будет скучать...")
            break
        else:
            print("Неизвестная команда!")

        pet.tick()  # время идёт

    print(f"\nИгра окончена! {pet.name} прожил {pet.age} ходов.")

import random class Pet: def __init__(self, name): self.name = name self.hp = 100 self.hunger = 30 self.sick = False self.medicine = 2 # аптечки def tick(self): self.hunger = min(100, self.hunger + 15) if self.hunger >= 80: self.hp -= 10 # Шанс заболеть if not self.sick and random.random() < 0.2: self.sick = True print(f"🤧 {self.name} заболел!") if self.sick: self.hp -= 5 def heal(self): if not self.sick: print(f"💊 {self.name} здоров!") elif self.medicine > 0: self.sick = False self.medicine -= 1 self.hp = min(100, self.hp + 20) print(f"💊 {self.name} вылечен! Аптечек: {self.medicine}") else: print(f"❌ Нет аптечек!") def eat(self): self.hunger = max(0, self.hunger - 30) print(f"🍖 {self.name} поел. Голод: {self.hunger}") def status(self): sick = "🤧больной" if self.sick else "😊здоров" print(f"🐾 {self.name} | HP:{self.hp} | голод:{self.hunger} | {sick} | 💊{self.medicine}") random.seed(3) pet = Pet("Пиксель") print("=== 5 дней Тамагочи 2.0 ===") for day in range(1, 6): print(f"\nДень {day}:") pet.status() if day == 3: pet.heal() # лечим на 3-й день else: pet.eat() pet.tick()

Магия __str__: питомец говорит о себе

Зачем __str__?

  • print(pet) → вызывает pet.__str__()
  • Удобно при отладке — сразу видно состояние
  • f-строки: f"Питомец: {pet}" — тоже работает
  • Без __str__ → <Pet object at 0x...> — некрасиво
Python
class Pet: def __init__(self, name): self.name = name self.hunger = 4 self.happiness = 8 self.energy = 7 def __str__(self): def bar(v, mx=10): filled = round(v / mx * 8) return "█" * filled + "░" * (8 - filled) return ( f"🐾 {self.name}\n" f" Голод: [{bar(self.hunger)}] {self.hunger}/10\n" f" Счастье: [{bar(self.happiness)}] {self.happiness}/10\n" f" Энергия: [{bar(self.energy)}] {self.energy}/10" ) p = Pet("Пушок") print(p) # вызывает __str__ print(f"Мой питомец:\n{p}") # тоже работает

Состояние питомца → словарь

Python
class Pet: def __init__(self, name, hunger=3, happiness=7, energy=8, age=0): self.name = name self.hunger = hunger self.happiness = happiness self.energy = energy self.age = age def as_dict(self): return { "name": self.name, "hunger": self.hunger, "happiness": self.happiness, "energy": self.energy, "age": self.age, } @classmethod def from_dict(cls, data): return cls(**data) # распаковываем словарь как аргументы def __str__(self): return f"🐾 {self.name} | голод:{self.hunger} | счастье:{self.happiness} | возраст:{self.age}" # Сохраняем питомца p1 = Pet("Пушок") print("Создали:", p1) saved = p1.as_dict() print("Словарь:", saved) # Загружаем питомца из словаря p2 = Pet.from_dict(saved) print("Загружен:", p2) # Изменяем и снова сохраняем p2.age = 5 p2.hunger = 6 print("После изменений:", p2.as_dict())

Инвентарь: список еды для питомца

Python
class Pet: def __init__(self, name): self.name = name self.hunger = 5 self.inventory = [] # список еды def buy_food(self, food): self.inventory.append(food) print(f"🛒 Куплено: {food}. Инвентарь: {self.inventory}") def feed(self, food=None): if food is None: if self.inventory: food = self.inventory.pop(0) # берём первую еду else: print(f"❌ {self.name}: нет еды!") return else: if food not in self.inventory: print(f"❌ {food} нет в инвентаре!") return self.inventory.remove(food) self.hunger = max(0, self.hunger - 3) print(f"🍖 {self.name} съел {food}! Голод: {self.hunger}") def status(self): print(f"🐾 {self.name}: голод={self.hunger} | инвентарь: {self.inventory}") cat = Pet("Мурка") cat.buy_food("рыба") cat.buy_food("молоко") cat.buy_food("мясо") cat.status() cat.feed("рыба") cat.feed() # берёт следующую (молоко) cat.feed("торт") # нет такой еды! cat.status()

Частые ошибки при написании Pet

❌ Общий список для всех объектов

class Pet: inventory = [] # ❌ общий для ВСЕХ питомцев! def __init__(self, name): self.name = name def add_food(self, food): self.inventory.append(food) p1 = Pet("Мурка") p2 = Pet("Шарик") p1.add_food("рыба") print(p2.inventory) # ["рыба"] — у Шарика тоже появилась!

✓ Свой список в __init__

class Pet: def __init__(self, name): self.name = name self.hunger = 3 self.happiness = 7 self.energy = 8 self.age = 0 def _clamp(self, v): return max(0, min(10, v)) def feed(self): self.hunger = self._clamp(self.hunger - 3) print(f"🍖 {self.name} поел! Голод: {self.hunger}") def tick(self): self.hunger = self._clamp(self.hunger + 1) self.happiness = self._clamp(self.happiness - 1) self.age += 1 class Dragon(Pet): def __init__(self, name): super().__init__(name) self.fire_level = 5 # уровень огня 0-10 def breathe_fire(self): if self.fire_level < 2: print(f"🔥 {self.name}: огонь закончился...") else: self.fire_level -= 2 self.happiness = self._clamp(self.happiness + 3) print(f"🔥 {self.name} выдохнул огонь! Огонь: {self.fire_level}") def fly(self): if self.energy < 3: print(f"😴 {self.name} слишком устал летать") else: self.energy = self._clamp(self.energy - 3) self.happiness = self._clamp(self.happiness + 4) print(f"🐉 {self.name} полетел! Счастье: {self.happiness}") def play(self): self.breathe_fire() # дракон играет огнём! drago = Dragon("Дракко") drago.fly() drago.breathe_fire() drago.breathe_fire() drago.feed() drago.tick() print(f"\n{drago.name}: счастье={drago.happiness}, огонь={drago.fire_level}, возраст={drago.age}")
