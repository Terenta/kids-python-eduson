Python — симуляция порядка слоёв
# Симулируем порядок draw() layers = [] def fill(color): layers.append(f" screen.fill('{color}')") def filled_rect(x, y, w, h, color): layers.append(f" filled_rect({x},{y},{w},{h}, '{color}')") def filled_circle(cx, cy, r, color): layers.append(f" filled_circle({cx},{cy}, r={r}, '{color}')") def draw_actor(name): layers.append(f" {name}.draw()") # Порядок рисования: fill('skyblue') filled_rect(0, 530, 800, 70, 'brown') for cx in [180, 220, 260]: filled_circle(cx, 80, 30, 'white') draw_actor('basket') draw_actor('apple') print("Порядок слоёв draw():") for i, step in enumerate(layers, 1): print(f"{i}.{step}")

Корзинка следит за мышью

Событие движения

on_mouse_move(pos)
Вызывается при каждом движении мыши

Параметр pos

pos = (x, y) — координаты курсора в пикселях. pos[0] = X, pos[1] = Y

Событие клика

on_mouse_down(pos, button)
button: mouse.LEFT / mouse.RIGHT

PyGame Zero вызывает on_mouse_move автоматически — не нужно вызывать вручную из update()!

Корзинка по Y не двигается — Y фиксирован на 540, меняем только X в зависимости от позиции курсора

Функция on_mouse_move

WIDTH = 800
HEIGHT = 600
TITLE = "Поймай яблоко!"

basket = Actor('basket')
basket.pos = (400, 540)

apple = Actor('apple')
apple.pos = (400, 50)

def draw():
    screen.fill('skyblue')
    basket.draw()
    apple.draw()

def update():
    # яблоко падает
    apple.y += 3
    if apple.y > HEIGHT:
        apple.y = 0

# ← PyGame Zero вызывает сам!
def on_mouse_move(pos):
    # Двигаем корзинку по X
    basket.x = pos[0]

    # Ограничиваем края экрана
    half = basket.width // 2
    basket.x = max(half,
                   min(WIDTH - half,
                       basket.x))

Как работает clamping

# max(half, x) — не меньше half
# min(WIDTH-half, x) — не больше
# вместе: держим x в диапазоне
x = max(50, min(750, x))
# x всегда от 50 до 750

Корзинка по Y не двигается — Y фиксирован на 540, меняем только X

# Клик мышью
def on_mouse_down(pos, button):
    if button == mouse.LEFT:
        print(f"Клик в {pos}")

        # Пример: создать новое яблоко
        apple.x = random.randint(
            50, WIDTH - 50
        )
        apple.y = 0

    if button == mouse.RIGHT:
        print("Правая кнопка!")

# Кнопка мыши отпущена
def on_mouse_up(pos, button):
    print(f"Отпустили в {pos}")

Идея расширения

Клик — «ловушка-сеть»: увеличить корзинку на 2 секунды. Используем clock.schedule()!

Временные эффекты: clock.schedule

import random

basket = Actor('basket')
basket.pos = (400, 540)
normal_width = 80
big_width = 160

def on_mouse_down(pos, button):
    if button == mouse.LEFT:
        # Расширяем корзинку
        basket.image = 'basket_big'

        # Возвращаем через 2 секунды
        clock.schedule(
            reset_basket,  # ← функция
            2.0            # ← секунды
        )

def reset_basket():
    # Возвращаем обычную корзинку
    basket.image = 'basket'

Синтаксис clock

# Вызвать через N секунд
clock.schedule(func, N)

# Повторять каждые N секунд
clock.schedule_interval(func, N)

# Отменить запланированное
clock.unschedule(func)

Экран Game Over

WIDTH = 800; HEIGHT = 600
basket = Actor('basket')
basket.pos = (400, 550)
apple = Actor('apple')
apple.pos = (400, 0)
score = 0
lives = 3
game_state = 'playing'  # или 'game_over'

def draw():
    screen.fill((135, 206, 235))

    if game_state == 'playing':
        basket.draw()
        apple.draw()
        screen.draw.text(
            f'Счёт: {score}  Жизни: {lives}',
            (10, 10), color='white', fontsize=28
        )
    elif game_state == 'game_over':
        screen.draw.text('GAME OVER', (250, 200),
            color='red', fontsize=72)
        screen.draw.text(f'Счёт: {score}',
            (320, 300), color='white', fontsize=40)
        screen.draw.text('Нажми ПРОБЕЛ',
            (280, 380), color='yellow', fontsize=28)

def update():
    global lives, game_state, apple
    if game_state != 'playing':
        return
    apple.y += 3
    if apple.y > HEIGHT:
        lives -= 1
        apple.y = 0
        if lives <= 0:
            game_state = 'game_over'

Restart по пробелу

def on_key_down(key):
    global game_state, score, lives
    if key == keys.SPACE:
        if game_state == 'game_over':
            score = 0
            lives = 3
            game_state = 'playing'

Три состояния

  • 'playing' — игра идёт
  • 'game_over' — жизни кончились
  • 'win' — победа (score > 20)

Чего не делать

Логика в draw()

# Плохо
def draw():
    apple.y += 3  # логика!

draw() — только рисование! Движение — в update()

Забыть global

# Плохо
score = 0
def update():
    score += 1  # ошибка!

# Хорошо
def update():
    global score
    score += 1

Изменяешь глобальную переменную — объяви global

PNG не найден

# Плохо — файл в корне
basket.png

# Правильно
images/basket.png
# рядом с game.py

Папка images/ должна лежать рядом с game.py

Правило: draw() — только screen.fill и .draw(). update() — вся логика. on_mouse_move() — только позиция.

import random WIDTH = 800 HEIGHT = 600 score = 0 lives = 3 speed = 3 # Позиции (симуляция Actor) apple_x = random.randint(50, 750) apple_y = 0 basket_x = 400 BASKET_HALF = 40 print("=== Симуляция 20 кадров игры ===") for frame in range(20): # Движение яблока apple_y += speed # Clamping корзинки (мышь = случайный x) mouse_x = random.randint(0, WIDTH) basket_x = max(BASKET_HALF, min(WIDTH - BASKET_HALF, mouse_x)) # Коллизия (упрощённая) if abs(apple_x - basket_x) < BASKET_HALF and apple_y >= HEIGHT - 70: score += 1 speed = 3 + score * 0.3 apple_x = random.randint(50, 750) apple_y = 0 print(f" Кадр {frame+1:2d}: ПОЙМАЛ! score={score}, speed={speed:.1f}") elif apple_y > HEIGHT: lives -= 1 apple_x = random.randint(50, 750) apple_y = 0 print(f" Кадр {frame+1:2d}: Промазал. Жизни: {lives}") if lives <= 0: print("GAME OVER!") break print(f"\nИтог: score={score}, lives={lives}")
