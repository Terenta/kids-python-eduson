Типичные ошибки в ДЗ

  • actor.image вместо images[frame] — не даёт смены кадров
  • Нет ANIMATE_SPEED — анимация слишком быстрая
  • frame без % len(images) — IndexError при выходе за список
  • global animation_frame — обязательно если меняешь в update()

Эталонное решение — спрайт с анимацией

WIDTH = 800; HEIGHT = 600

# Список кадров анимации
images = ['hero1', 'hero2', 'hero3', 'hero2']
ANIMATE_SPEED = 8   # кадров до смены

hero = Actor(images[0])
hero.pos = (400, 300)
animation_frame = 0
tick = 0

def draw():
    screen.fill('darkblue')
    hero.draw()
    screen.draw.text(
        f'Кадр: {animation_frame}',
        (10, 10), color='white', fontsize=24
    )

def update():
    global animation_frame, tick
    tick += 1
    if tick % ANIMATE_SPEED == 0:
        animation_frame = (animation_frame + 1) % len(images)
        hero.image = images[animation_frame]

import pgzrun; pgzrun.go()

Ключевые паттерны

  • tick % ANIMATE_SPEED == 0 — замедляет анимацию
  • % len(images) — циклически возвращает к началу
  • hero.image = images[frame] — меняет спрайт
tick=0 frame=0 hero1
tick=8 frame=1 hero2
tick=16 frame=2 hero3
tick=24 frame=3 hero2
tick=32 frame=0 hero1 ← цикл!

Система координат в PyGame Zero

Правила движения

# ← влево: X уменьшается
player.x -= 5

# → вправо: X увеличивается
player.x += 5

# ↑ вверх: Y уменьшается!

# ↓ вниз: Y увеличивается
player.y += 5
Запомни: в играх Y растёт вниз. Это отличается от математики!

Персонаж не вылетает — boundary checking

Без проверки — улетает!

def update():
    if keyboard.right:
        basket.x += 5
    # basket.x = 9999!

С проверкой — остаётся на экране

def update():
    if keyboard.right:
        basket.x += 5
    # ограничиваем края
    if basket.right > WIDTH:
        basket.right = WIDTH
    if basket.left < 0:
        basket.left = 0
# Способ 2: clamp (короче)
def update():
    speed = 5
    if keyboard.left:
        basket.x -= speed
    if keyboard.right:
        basket.x += speed

    # Один clamp для обоих краёв
    half = basket.width // 2
    basket.x = max(half,
                   min(WIDTH - half,
                       basket.x))

# Способ 3: .left/.right напрямую
if basket.left < 0:   basket.left = 0
if basket.right > WIDTH: basket.right = WIDTH
basket.left и basket.right — встроенные свойства Actor. Они учитывают размер спрайта!

keyboard vs on_key_down

  • keyboard.left — True пока зажата. Используй в update() для движения.
  • on_key_down(key) — срабатывает один раз. Используй для прыжка, выстрела, паузы.

Делаем движение живым

WIDTH = 800; HEIGHT = 600

basket = Actor('basket')
basket.pos = (400, 540)
speed = 0         # начальная скорость
MAX_SPEED = 12    # ограничение

def update():
    global speed

    # Ускорение при нажатии
    if keyboard.right:
        speed += 1
    elif keyboard.left:
        speed -= 1
    else:
        # Торможение (трение)
        speed *= 0.85

    # Ограничиваем скорость
    speed = max(-MAX_SPEED,
                min(MAX_SPEED, speed))

    basket.x += speed

    # Границы
    basket.x = max(50, min(750, basket.x))
speed *= 0.85 — умножение на 0.85 каждый кадр создаёт плавное торможение

import random

WIDTH = 800; HEIGHT = 600; TITLE = "Поймай яблоко!"

basket = Actor('basket'); basket.pos = (400, 550)
apple = Actor('apple'); apple.pos = (random.randint(50, 750), 0)
score = 0; basket_speed = 0; fall_speed = 3; paused = False

def draw():
    screen.fill((135, 206, 235))
    screen.draw.filled_rect(Rect(0,530,800,70), (34,139,34))
    basket.draw(); apple.draw()
    screen.draw.text(f'Счёт: {score}', (10,10), color='white', fontsize=36)
    if paused:
        screen.draw.text('ПАУЗА', center=(400,300), color='yellow', fontsize=72)

def update():
    global score, basket_speed, fall_speed
    if paused: return

    # Клавиатурное управление
    if keyboard.right: basket_speed = min(12, basket_speed + 1.5)
    elif keyboard.left: basket_speed = max(-12, basket_speed - 1.5)
    else: basket_speed *= 0.8

    basket.x += basket_speed
    basket.x = max(50, min(750, basket.x))

    apple.y += fall_speed
    if apple.colliderect(basket):
        score += 1; fall_speed = 3 + score * 0.4
        apple.pos = (random.randint(50,750), 0)
    if apple.y > HEIGHT:
        apple.pos = (random.randint(50,750), 0)

def on_key_down(key):
    global paused
    if key == keys.ESCAPE or key == keys.SPACE:
        paused = not paused

def on_mouse_move(pos):  # бонус: мышь тоже работает!
    basket.x = max(50, min(750, pos[0]))

Движение во всех направлениях

WIDTH = 800; HEIGHT = 600
player = Actor('player')
player.pos = (400, 300)
SPEED = 5

def draw():
    screen.fill('darkgreen')
    player.draw()

def update():
    # Стрелки или WASD
    if keyboard.left  or keyboard.a:
        player.x -= SPEED
    if keyboard.right or keyboard.d:
        player.x += SPEED
    if keyboard.up    or keyboard.w:
        player.y -= SPEED  # Y вверх!
    if keyboard.down  or keyboard.s:
        player.y += SPEED

    # Границы по всем краям
    player.left   = max(0, player.left)
    player.right  = min(WIDTH, player.right)
    player.top    = max(0, player.top)
    player.bottom = min(HEIGHT, player.bottom)
Используй независимые if (не elif!) — тогда диагональное движение работает.

Словарь урока

keyboard
Объект PyGame Zero — состояние клавиш прямо сейчас. keyboard.left = True пока зажата
update()
Вызывается ~60 раз/сек. Здесь проверяем keyboard и двигаем объекты каждый кадр
on_key_down(key)
Событие — срабатывает один раз при нажатии. Для выстрела, паузы, скачка
clamping
Ограничение в диапазоне: max(min_val, min(max_val, x)) — корзинка не уходит за экран
инерция
speed нарастает при нажатии и умножается на 0.85 при отпускании — плавное торможение
colliderect()
Метод Actor. Проверяет пересечение двух прямоугольных bounding box-ов
global
Ключевое слово: нужно внутри функции, если изменяешь переменную из внешней области видимости
