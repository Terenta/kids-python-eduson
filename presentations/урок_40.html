def reset_game():
    global score, lives, fall_speed, basket_speed, state
    score = 0; lives = 3; fall_speed = 3; basket_speed = 0
    state = 'playing'
    apple.pos = (random.randint(50,750), 0); basket.pos = (400,550)

def update():
    global score, best_score, lives, fall_speed, basket_speed, state
    if state != 'playing': return

    # Клавиатура
    if keyboard.right: basket_speed = min(12, basket_speed + 1.5)
    elif keyboard.left:  basket_speed = max(-12, basket_speed - 1.5)
    else:               basket_speed *= 0.8
    basket.x += basket_speed
    basket.x = max(50, min(750, basket.x))

    # Яблоко падает
    apple.y += fall_speed
    if apple.colliderect(basket):   # поймали!
        score += 1; best_score = max(score, best_score)
        fall_speed = min(20, 3 + score * 0.5)
        apple.pos = (random.randint(50,750), 0)
    elif apple.y > HEIGHT:           # пропустили
        lives -= 1
        apple.pos = (random.randint(50,750), 0)
        if lives <= 0: state = 'game_over'

def on_mouse_move(pos):
    if state == 'playing':
        basket.x = max(50, min(750, pos[0]))

def on_key_down(key):
    global state
    if key == keys.R: reset_game()
    if key == keys.SPACE:
        if state == 'playing': state = 'paused'
        elif state == 'paused': state = 'playing'

import random

# Список яблок вместо одного
apples = []

def spawn_apple():
    # Создаём новое яблоко
    a = Actor('apple')
    a.x = random.randint(50, 750)
    a.y = -30
    apples.append(a)

# Первое яблоко при старте
spawn_apple()

def draw():
    screen.fill('skyblue')
    basket.draw()
    for a in apples:   # рисуем все
        a.draw()

def update():
    global score, lives
    for a in apples[:]:  # копия!
        a.y += fall_speed
        if a.colliderect(basket):
            score += 1; apples.remove(a)
            spawn_apple()
        elif a.y > HEIGHT:
            lives -= 1; apples.remove(a)
            spawn_apple()
