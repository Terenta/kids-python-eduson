Прыжок с гравитацией

Как работает физика в 15 строках кода

Принцип:

  • vy = вертикальная скорость (пикс/кадр)
  • Прыжок: vy = -12 (вверх — отрицательно)
  • Каждый кадр: vy += GRAVITY → тормозит
  • Позиция: y += vy
  • Земля: если y >= 300 → останавливаем

GRAVITY = 0.6 GROUND = 300 y = GROUND vy = -12 # прыжок! print("Симуляция прыжка:") for frame in range(50): vy += GRAVITY y += vy if y >= GROUND: y = GROUND vy = 0 print(f"Кадр {frame:2}: ПРИЗЕМЛЕНИЕ!") break height = GROUND - y bar = "#" * int(height / 8) print(f"Кадр {frame:2}: h={height:.0f} {bar}")

import random

WIDTH, HEIGHT = 800, 400
GROUND = 340
GRAVITY = 0.6
RUN_FRAMES = ['dino_run1', 'dino_run2']

game_state = 'menu'
score = 0
speed = 5
tick = 0

dino = Actor('dino_run1', (120, GROUND))
dino.vy = 0
cacti = []

def spawn_cactus():
    c = Actor(random.choice(['cactus_s', 'cactus_m']), (WIDTH + 50, GROUND))
    cacti.append(c)

def draw():
    screen.fill((135, 206, 235))
    screen.draw.filled_rect(Rect(0, GROUND+40, WIDTH, 80), (80, 140, 50))
    if game_state == 'playing':
        dino.draw()
        for c in cacti: c.draw()
        screen.draw.text(f"Score: {score}", (10,10), color='white', fontsize=32)
    elif game_state == 'menu':
        screen.draw.text("DINO RUN", center=(WIDTH//2, 150), fontsize=64, color='white')
        screen.draw.text("Нажми ПРОБЕЛ", center=(WIDTH//2, 250), fontsize=28, color=(200,200,200))
    elif game_state == 'gameover':
        screen.draw.text(f"Счёт: {score}", center=(WIDTH//2, 180), fontsize=42, color='white')
        screen.draw.text("ПРОБЕЛ — заново", center=(WIDTH//2, 240), fontsize=26, color=(200,200,200))

def update():
    global game_state, score, speed, tick
    if game_state != 'playing': return
    tick += 1; score += 1
    if tick % 3 == 0: speed = 5 + score // 200
    dino.vy += GRAVITY; dino.y += dino.vy
    if dino.y >= GROUND: dino.y = GROUND; dino.vy = 0
    dino.image = RUN_FRAMES[(tick // 6) % 2] if dino.y >= GROUND else 'dino_jump'
    for c in cacti[:]:
        c.x -= speed
        if c.x < -60: cacti.remove(c)
        if dino.colliderect(c): game_state = 'gameover'; sounds.hit.play()

def on_key_down(key):
    global game_state, score, cacti, tick
    if key == keys.SPACE:
        if game_state == 'menu': game_state = 'playing'; clock.schedule_interval(spawn_cactus, 1.5)
        elif game_state == 'gameover': game_state = 'playing'; score = 0; cacti.clear(); tick = 0; dino.y = GROUND
        elif game_state == 'playing' and dino.y >= GROUND: dino.vy = -12; sounds.jump.play()

Математика тяги: vx += math.cos(math.radians(ship.angle)) * THRUST — формула движения в направлении угла.

math.cos/sin для движения

Превращаем угол корабля в вектор скорости

Как это работает:

  • Угол корабля — ship.angle в градусах
  • math.radians(угол) → переводим в радианы
  • cos(рад) → горизонтальная составляющая X
  • sin(рад) → вертикальная составляющая Y
  • Тяга: vx += cos(рад) * THRUST
Python
import math THRUST = 1.0 # Попробуй: 0, 90, 180, 45, 270 angles = [0, 45, 90, 180, 270] print("Угол → направление движения:") for angle in angles: rad = math.radians(angle) vx = math.cos(rad) * THRUST vy = -math.sin(rad) * THRUST # -sin: y растёт вниз direction = "" if vx > 0.5: direction += "→" if vx < -0.5: direction += "←" if vy < -0.5: direction += "↑" if vy > 0.5: direction += "↓" print(f" {angle:3}° {direction:3} vx={vx:+.2f}, vy={vy:+.2f}")

Скелет Asteroids

Вращение + тяга + выстрелы — классическая механика

import math, random

WIDTH, HEIGHT = 800, 600
game_state = 'menu'
score = 0

ship = Actor('ship', (WIDTH//2, HEIGHT//2))
ship.vx = ship.vy = 0
THRUST = 0.3
ROTATE_SPEED = 4

asteroids = []
bullets = []

def spawn_asteroids(n=4):
    for _ in range(n):
        a = Actor('asteroid_big', (random.randint(0,WIDTH), random.randint(0,HEIGHT)))
        a.vx, a.vy = random.uniform(-2,2), random.uniform(-2,2)
        asteroids.append(a)

def update():
    global game_state, score
    if game_state != 'playing': return
    if keyboard.left:  ship.angle += ROTATE_SPEED
    if keyboard.right: ship.angle -= ROTATE_SPEED
    if keyboard.up:
        rad = math.radians(ship.angle)
        ship.vx += math.cos(rad) * THRUST
        ship.vy -= math.sin(rad) * THRUST
    ship.x = (ship.x + ship.vx) % WIDTH
    ship.y = (ship.y + ship.vy) % HEIGHT
    for b in bullets[:]:
        b.x += b.vx; b.y += b.vy
        if not (0 < b.x < WIDTH and 0 < b.y < HEIGHT): bullets.remove(b); continue
        for a in asteroids[:]:
            if b.colliderect(a):
                bullets.remove(b); asteroids.remove(a); score += 10
                sounds.boom.play(); break
    for a in asteroids:
        a.x = (a.x + a.vx) % WIDTH; a.y = (a.y + a.vy) % HEIGHT
        if ship.colliderect(a): game_state = 'gameover'; sounds.hit.play()

def on_key_down(key):
    if key == keys.SPACE and game_state == 'playing':
        rad = math.radians(ship.angle)
        b = Actor('bullet', ship.pos)
        b.vx = math.cos(rad) * 8; b.vy = -math.sin(rad) * 8
        bullets.append(b); sounds.shoot.play()

Паттерн: State Machine

Система состояний — основа любой игры

State Machine — программа, которая всегда знает, в каком состоянии находится, и как переходить между ними.
  • В любой момент — одно активное состояние
  • Переходы — только по событиям (клавиша, коллизия)
  • Каждое состояние — своя логика draw() и update()

Этот паттерн используют в Unity, Unreal Engine — и во всех серьёзных Python-проектах.

State Machine в коде

Python
class GameStateMachine: def __init__(self): self.state = 'menu' self.transitions = { 'menu': ['playing'], 'playing': ['paused', 'gameover'], 'paused': ['playing'], 'gameover': ['menu'], } def go(self, new_state): allowed = self.transitions.get(self.state, []) if new_state in allowed: print(f" ✓ {self.state} → {new_state}") self.state = new_state else: print(f" ✗ Нельзя: {self.state} → {new_state}") game = GameStateMachine() print(f"Старт: {game.state}") game.go('playing') # ок game.go('paused') # ок game.go('gameover') # нельзя из paused! game.go('playing') # ок — снять паузу game.go('gameover') # ок game.go('menu') # ок print(f"Финал: {game.state}")

Как PyGame Zero определяет столкновения

Принцип:

Каждый Actor — прямоугольник (Rect). colliderect(b) проверяет: пересекаются ли два прямоугольника?

Для круглых объектов уменьши хитбокс — игрок не заметит, но будет доволен!

Python
class Rect: def __init__(self, x, y, w, h): self.x, self.y = x, y self.w, self.h = w, h def inflate(self, dw, dh): return Rect(self.x - dw//2, self.y - dh//2, self.w + dw, self.h + dh) def colliderect(self, other): return (self.x < other.x + other.w and self.x + self.w > other.x and self.y < other.y + other.h and self.y + self.h > other.y) dino = Rect(100, 100, 40, 40) cactus = Rect(120, 90, 30, 60) hit = dino.colliderect(cactus) print(f"Обычная коллизия: {hit}") soft = dino.inflate(-16, -16) hit2 = soft.colliderect(cactus) print(f"Мягкая коллизия: {hit2}") # Подвинь кактус подальше: far = Rect(300, 100, 30, 60) print(f"Далеко: {dino.colliderect(far)}")

Хай-скор в файл

Рекорд, который переживёт перезапуск игры

Без файла — рекорд теряется при закрытии. С файлом — живёт вечно.
  1. При запуске: если файл есть → загрузить рекорд
  2. При Game Over: если score > highscore → записать в файл
  3. На экране Game Over: показать «Рекорд: N»

Схема:

# При старте:
highscore = load_hs()

# В game over:
if score > highscore:
  highscore = score
  save_hs(highscore)

# На экране:
f"Рекорд: {highscore}"

os.path.exists() — проверяет, есть ли файл. Защита от ошибки при первом запуске.

Код: save / load highscore

Python
import os HS_FILE = 'highscore.txt' def load_highscore(): if os.path.exists(HS_FILE): with open(HS_FILE, 'r') as f: return int(f.read().strip()) return 0 # первый запуск — рекорд 0 def save_highscore(score): with open(HS_FILE, 'w') as f: f.write(str(score)) # Симуляция игровых сессий highscore = load_highscore() print(f"Загружен рекорд: {highscore}") sessions = [120, 450, 380, 610, 200] for i, score in enumerate(sessions, 1): if score > highscore: highscore = score save_highscore(highscore) print(f"Сессия {i}: {score} очков — НОВЫЙ РЕКОРД!") else: print(f"Сессия {i}: {score} очков (рекорд {highscore})") print(f"\nФинальный рекорд: {highscore}")
